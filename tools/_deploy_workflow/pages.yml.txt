# Publica o site estático no GitHub Pages (Settings → Pages → Source: "GitHub Actions").
# Publica a raiz do repositório, exceto pastas internas e arquivos de controle.
name: Publicar no GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Obter o código
        uses: actions/checkout@v4

      - name: Configurar o Pages
        uses: actions/configure-pages@v5

      - name: Montar a pasta publicada (sem pastas internas)
        run: |
          mkdir -p _site
          rsync -a ./ _site/ \
            --exclude '.git/' \
            --exclude '.github/' \
            --exclude '_site/' \
            --exclude '_amostra/' \
            --exclude '_relatorios/' \
            --exclude 'tools/' \
            --exclude 'data/pendencias.json' \
            --exclude 'data/manifesto.csv' \
            --exclude '.gitignore'
          test -f _site/index.html
          test -f _site/.nojekyll

      - name: Enviar artefato do Pages
        uses: actions/upload-pages-artifact@v3
        with:
          path: _site

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Publicar
        id: deployment
        uses: actions/deploy-pages@v4
